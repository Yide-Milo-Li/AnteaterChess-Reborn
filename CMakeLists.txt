cmake_minimum_required(VERSION 3.21)
project(AnteaterChess LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(AC_BUILD_DESKTOP "Build Qt desktop and adapter tests" ON)
option(AC_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
option(AC_WINDOWS_CONSOLE "Attach desktop to a console" OFF)
include(CTest)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
file(READ "${CMAKE_SOURCE_DIR}/VERSION" AC_VERSION)
string(STRIP "${AC_VERSION}" AC_VERSION)
add_compile_definitions(_POSIX_C_SOURCE=200809L AC_VERSION="${AC_VERSION}")
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    add_compile_options(-Wall -Wextra -Werror -UNDEBUG)
    if(AC_SANITIZE)
        add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
        add_link_options(-fsanitize=address,undefined)
    endif()
endif()
foreach(module rules ai session)
    file(GLOB sources CONFIGURE_DEPENDS "src/${module}/*.c")
    add_library(ac_${module} STATIC ${sources})
    target_include_directories(ac_${module} PUBLIC include)
    set_target_properties(ac_${module} PROPERTIES ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
endforeach()
target_link_libraries(ac_ai PUBLIC ac_rules)
target_link_libraries(ac_session PUBLIC ac_ai)
add_custom_target(headless DEPENDS ac_session)
file(GLOB core_tests CONFIGURE_DEPENDS tests/rules/test_*.c tests/session/test_*.c tests/ai/test_*.c)
foreach(source ${core_tests})
    get_filename_component(name ${source} NAME_WE)
    get_filename_component(directory ${source} DIRECTORY)
    get_filename_component(category ${directory} NAME)
    add_executable(${name} ${source})
    target_link_libraries(${name} PRIVATE ac_session)
    set_target_properties(${name} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/tests/${category}")
    add_test(NAME ${name} COMMAND ${name})
    set_tests_properties(${name} PROPERTIES LABELS "core;${category}")
endforeach()
add_executable(test_cxx_linkage tests/core/test_cxx_linkage.cpp)
target_link_libraries(test_cxx_linkage PRIVATE ac_session)
add_test(NAME test_cxx_linkage COMMAND test_cxx_linkage)
set_tests_properties(test_cxx_linkage PROPERTIES LABELS core)
add_executable(ai_reference tools/dev/ai_reference.c)
target_include_directories(ai_reference PRIVATE src/ai)
target_link_libraries(ai_reference PRIVATE ac_session)
add_test(NAME test_ai_reference COMMAND ${Python3_EXECUTABLE} "${CMAKE_SOURCE_DIR}/tools/dev/compare_ai_reference.py"
    $<TARGET_FILE:ai_reference> "${CMAKE_SOURCE_DIR}/tests/fixtures/ai-evaluation-baseline.txt")
set_tests_properties(test_ai_reference PROPERTIES LABELS "core;ai")
add_executable(ac_benchmark tools/dev/benchmark.c)
target_link_libraries(ac_benchmark PRIVATE ac_session)
add_custom_target(check COMMAND ${Python3_EXECUTABLE} "${CMAKE_SOURCE_DIR}/tools/dev/check.py")
if(AC_BUILD_DESKTOP)
    find_package(Qt6 6.4.2 REQUIRED COMPONENTS Core Gui Qml Quick QuickControls2 Svg Test QuickTest)
    set(CMAKE_AUTOMOC ON)
    set(CMAKE_AUTORCC ON)
    add_library(ac_desktop STATIC
        apps/qt/runtime/runtime.cpp apps/qt/runtime/runtime.h
        apps/qt/models/board_model.cpp apps/qt/models/board_model.h
        apps/qt/models/history_model.cpp apps/qt/models/history_model.h
        apps/qt/async/search_jobs.cpp apps/qt/async/search_jobs.h
        apps/qt/app/session_adapter.cpp apps/qt/app/session_adapter.h)
    target_include_directories(ac_desktop PUBLIC apps/qt)
    target_link_libraries(ac_desktop PUBLIC ac_session Qt6::Core Qt6::Gui Qt6::Quick Qt6::Qml Qt6::Svg)
    add_library(ac_resources OBJECT assets/resources.qrc apps/qt/qml/qml.qrc)
    target_link_libraries(ac_resources PRIVATE Qt6::Core)
    add_executable(anteater-chess apps/qt/main.cpp $<TARGET_OBJECTS:ac_resources>)
    target_link_libraries(anteater-chess PRIVATE ac_desktop Qt6::QuickControls2)
    set_target_properties(anteater-chess PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
    if(WIN32 AND NOT AC_WINDOWS_CONSOLE)
        set_target_properties(anteater-chess PROPERTIES WIN32_EXECUTABLE ON)
    endif()
    foreach(name runtime desktop)
        add_executable(test_qt_${name} tests/qt/test_${name}.cpp $<TARGET_OBJECTS:ac_resources>)
        target_link_libraries(test_qt_${name} PRIVATE ac_desktop Qt6::Test Qt6::QuickControls2)
        set_target_properties(test_qt_${name} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/tests/qt")
        add_test(NAME test_qt_${name} COMMAND test_qt_${name} -o -,txt)
        set_tests_properties(test_qt_${name} PROPERTIES TIMEOUT 90)
        if(name STREQUAL "runtime")
            set_tests_properties(test_qt_${name} PROPERTIES LABELS platform)
        else()
            set_tests_properties(test_qt_${name} PROPERTIES LABELS gui)
        endif()
    endforeach()
    add_executable(test_qml tests/qt/test_qml.cpp $<TARGET_OBJECTS:ac_resources>)
    target_link_libraries(test_qml PRIVATE ac_desktop Qt6::QuickTest Qt6::QuickControls2)
    target_compile_definitions(test_qml PRIVATE QUICK_TEST_SOURCE_DIR="${CMAKE_SOURCE_DIR}/tests/qt/qml")
    add_test(NAME test_qml COMMAND test_qml -o -,txt)
    set_tests_properties(test_qml PROPERTIES LABELS gui ENVIRONMENT "QT_QUICK_CONTROLS_STYLE=Basic")
endif()
