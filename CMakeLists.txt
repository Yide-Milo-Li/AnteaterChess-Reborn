cmake_minimum_required(VERSION 3.25)
project(AnteaterChess LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(AC_BUILD_DESKTOP "Build Qt desktop and adapter tests" ON)
option(AC_BUILD_DEV_TOOLS "Build benchmarks and development checks" OFF)
option(AC_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
option(AC_WINDOWS_CONSOLE "Attach desktop to a console" OFF)
option(AC_BUILD_DISTRIBUTION "Enable runtime installation and dependency receipts" OFF)
include(CTest)
if(AC_BUILD_DEV_TOOLS OR AC_BUILD_DISTRIBUTION)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
endif()
file(READ "${CMAKE_SOURCE_DIR}/VERSION" AC_VERSION)
string(STRIP "${AC_VERSION}" AC_VERSION)
add_compile_definitions(_POSIX_C_SOURCE=200809L AC_VERSION="${AC_VERSION}")
if(MSVC)
    if(NOT MSVC_VERSION EQUAL 1944 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
        message(FATAL_ERROR "Windows requires native v143 14.44 x64. Initialize tools/native/Enter-NativeEnvironment.ps1 first.")
    endif()
    add_compile_options(/W4 /EHsc /Zc:__cplusplus /utf-8 /UNDEBUG)
    add_compile_definitions(_CRT_SECURE_NO_WARNINGS)
    if(CMAKE_GENERATOR MATCHES "Ninja")
        include(cmake/MSVCDependencies.cmake)
    endif()
elseif(WIN32)
    message(FATAL_ERROR "Windows builds require native MSVC v143, not MinGW or MSYS2.")
elseif(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    add_compile_options(-Wall -Wextra -Werror -UNDEBUG)
    if(AC_SANITIZE)
        add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
        add_link_options(-fsanitize=address,undefined)
    endif()
endif()
foreach(module rules ai policy session)
    file(GLOB sources CONFIGURE_DEPENDS "src/${module}/*.cpp")
    add_library(ac_${module} STATIC ${sources})
    target_include_directories(ac_${module} PUBLIC include)
    set_target_properties(ac_${module} PROPERTIES ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
endforeach()
target_link_libraries(ac_ai PUBLIC ac_rules)
target_link_libraries(ac_session PUBLIC ac_rules ac_policy)
add_custom_target(headless DEPENDS ac_session)
if(BUILD_TESTING)
add_library(ac_test_diagnostics OBJECT tests/core/native_diagnostics.cpp)
file(GLOB core_tests CONFIGURE_DEPENDS tests/rules/test_*.cpp tests/session/test_*.cpp tests/ai/test_*.cpp)
foreach(source ${core_tests})
    get_filename_component(name ${source} NAME_WE)
    get_filename_component(directory ${source} DIRECTORY)
    get_filename_component(category ${directory} NAME)
    add_executable(${name} ${source})
    target_sources(${name} PRIVATE $<TARGET_OBJECTS:ac_test_diagnostics>)
    target_link_libraries(${name} PRIVATE ac_session ac_ai)
    set_target_properties(${name} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/tests/${category}")
    add_test(NAME ${name} COMMAND ${name})
    set_tests_properties(${name} PROPERTIES LABELS "core;${category}" TIMEOUT 120)
endforeach()
add_executable(test_cxx_linkage tests/core/test_cxx_linkage.cpp)
target_sources(test_cxx_linkage PRIVATE $<TARGET_OBJECTS:ac_test_diagnostics>)
target_link_libraries(test_cxx_linkage PRIVATE ac_session ac_ai)
add_test(NAME test_cxx_linkage COMMAND test_cxx_linkage)
set_tests_properties(test_cxx_linkage PROPERTIES LABELS core)
add_executable(ai_reference tools/dev/ai_reference.cpp)
target_sources(ai_reference PRIVATE $<TARGET_OBJECTS:ac_test_diagnostics>)
target_include_directories(ai_reference PRIVATE src/ai)
target_link_libraries(ai_reference PRIVATE ac_session ac_ai)
add_test(NAME test_ai_reference COMMAND ai_reference "${CMAKE_SOURCE_DIR}/tests/fixtures/ai-evaluation-baseline.txt")
set_tests_properties(test_ai_reference PROPERTIES LABELS "core;ai")
endif()
if(AC_BUILD_DEV_TOOLS)
add_executable(ac_benchmark tools/dev/benchmark.cpp)
target_link_libraries(ac_benchmark PRIVATE ac_session ac_ai)
add_custom_target(check COMMAND ${Python3_EXECUTABLE} "${CMAKE_SOURCE_DIR}/tools/dev/check.py")
endif()
if(AC_BUILD_DESKTOP)
    include(cmake/Distribution.cmake)
    add_subdirectory(apps/qt)
else()
    include(cmake/Distribution.cmake)
endif()
